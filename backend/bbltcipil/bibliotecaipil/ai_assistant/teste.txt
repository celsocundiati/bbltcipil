from celery import shared_task
from .services import gerar_resposta
from bibliotecaipil.events import emit_event



@shared_task
def processar_mensagem_task(pergunta, user_id, chat_id):
    from django.contrib.auth.models import User
    from ai_assistant.models import Chat  # ajustar ao teu projeto

    user = User.objects.get(id=user_id)
    chat = Chat.objects.get(id=chat_id)

    resposta = gerar_resposta(pergunta, user, chat)

    emit_event(
        "mensagem_processada",
        {
            "user_id": user.id,
            "chat_id": chat.id,
            "pergunta": pergunta,
            "resposta": resposta,
        },
    )

    return resposta

from django.db.models.signals import post_save
from django.dispatch import receiver
from .models import Livro
from bibliotecaipil.events import emit_event


@receiver(post_save, sender=Livro)
def livro_criado(sender, instance, created, **kwargs):
    if created:
        emit_event("livro_criado", {
            "livro_id": instance.id
        })
